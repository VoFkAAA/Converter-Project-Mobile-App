# 1. Клонировать репозиторий и перейти в папку mobile
git clone https://github.com/VoFkAAA/Converter-Project-Mobile-App
cd Converter-Project-Mobile-App/mobile

# 2. Установить зависимости Node.js
npm install

# 3. Установить Capacitor (если ещё не установлен в зависимостях)
npm install @capacitor/core @capacitor/cli @capacitor/android

# 4. Инициализировать Capacitor (создаст capacitor.config.json)
npx cap init "Конвертер даты" "com.qa.dateconverter" --web-dir=.

# 5. Добавить Android-платформу (создаст папку android)
npx cap add android

# 6. Скопировать веб-файлы (index.html) в Android-проект
npx cap copy

# 7. Перейти в папку Android и собрать APK
cd android
.\gradlew assembleDebug --no-daemon			# Windows
./gradlew assembleDebug --no-daemon			# Mac/Linux

# 8. Готовый APK находится здесь:
Converter-Project-Mobile-App/mobile/android/app/build/outputs/apk/debug/app-debug.apk
